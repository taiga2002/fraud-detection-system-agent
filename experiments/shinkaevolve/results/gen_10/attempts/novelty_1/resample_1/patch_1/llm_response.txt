def compute_fusion_score(isolation_forest_scores, account_risk_scores):
    """
    Combine isolation forest risk and account risk into a final suspicious-transfer score.

    Args:
        isolation_forest_scores: numpy array of anomaly scores from the Isolation Forest model
        account_risk_scores: numpy array of sender_account_risk_score

    Returns:
        fused_scores: numpy array of final risk scores (0-1)
        threshold: float, decision threshold for alerting
    """
    # Weighted fusion: isolation forest risk dominates, account risk amplifies
    w_forest = 0.80
    w_account = 0.20

    fused_scores = w_forest * np.clip(isolation_forest_scores, -1, 1) + w_account * account_risk_scores

    # Boost fused score when account risk is elevated
    high_account_risk = account_risk_scores > 0.3
    fused_scores = np.where(high_account_risk, np.clip(fused_scores * 1.2, 0, 1), fused_scores)

    # Adaptive threshold: use percentile-based to handle varying score ranges
    # Target ~5-8% alert rate (matching approximate fraud rate)
    threshold = float(np.percentile(fused_scores, 92))

    return fused_scores, threshold
