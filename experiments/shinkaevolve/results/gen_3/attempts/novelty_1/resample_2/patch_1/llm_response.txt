"""
Evolvable ML Pipeline for Japanese Banking Fraud Detection

This program is evolved by ShinkaEvolve. Code inside EVOLVE-BLOCK markers
can be mutated by the LLM. Everything outside is fixed infrastructure.

Outputs JSON metrics to stdout for the evaluator to parse.
"""

import sys
import os
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score, f1_score, precision_score, recall_score
import lightgbm as lgb

# ─── Fixed infrastructure: locate project root and load data ───
def _find_project_root():
    """Walk up from cwd and __file__ to find the project root with data/processed/."""
    candidates = [
        os.getcwd(),
        os.path.dirname(os.path.abspath(__file__)),
    ]
    for start in candidates:
        path = os.path.abspath(start)
        for _ in range(10):
            if os.path.isdir(os.path.join(path, 'data', 'processed')):
                return path
            parent = os.path.dirname(path)
            if parent == path:
                break
            path = parent
    raise FileNotFoundError("Cannot find project root with data/processed/ directory")

PROJECT_ROOT = _find_project_root()
DATA_DIR = os.path.join(PROJECT_ROOT, 'data', 'processed')


def load_data(seed=42):
    """Load and subsample transaction data for fast evaluation."""
    txn_path = os.path.join(DATA_DIR, 'transactions_with_features.csv')
    acc_path = os.path.join(DATA_DIR, 'accounts_with_risk.csv')

    txn_df = pd.read_csv(txn_path)
    acc_df = pd.read_csv(acc_path)

    # Subsample to 10K rows for speed (stratified by target)
    if len(txn_df) > 10000:
        txn_df, _ = train_test_split(
            txn_df, train_size=10000, random_state=seed,
            stratify=txn_df['is_laundering']
        )
        txn_df = txn_df.reset_index(drop=True)

    return txn_df, acc_df


class FeatureEngineer:
    """
    Creates interaction features from the base transaction columns.
    
    Attributes:
        df (pd.DataFrame): Input dataframe with transactional data
    """

    def __init__(self, df):
        self.df = df

    # EVOLVE-BLOCK-START
    def engineer_features(self):
        feats = pd.DataFrame(index=self.df.index)

        # IB enablement ratio: how recently IB was enabled relative to account age
        feats['ib_age_ratio'] = self.df['sender_ib_days_since_enabled'].fillna(9999) / (self.df['sender_account_age_days'] + 1)

        # Amount relative to account history
        feats['amount_to_mean_ratio'] = self.df['amount_jpy'] / (self.df['sender_mean_amount'] + 1)

        # Velocity intensity: volume per transaction in recent window
        feats['avg_recent_txn_amount'] = self.df['txn_volume_24h'] / (self.df['txn_count_24h'] + 1)

        # New account + high amount interaction
        feats['new_acct_high_amount'] = self.df['receiver_is_new'].fillna(0) * self.df['is_high_amount'].fillna(0)

        # Balance drain with first transfer
        feats['drain_first_transfer'] = self.df['balance_drain_ratio'].fillna(0) * self.df['first_transfer'].fillna(0)

        # Late night + high amount risk
        feats['late_night_high_amount'] = self.df['is_late_night'].fillna(0) * self.df['is_high_amount'].fillna(0)

        # Account risk amplified by velocity
        feats['risk_velocity'] = self.df['sender_account_risk_score'].fillna(0) * (self.df['txn_count_24h'].fillna(0) + 1)

        return feats
    # EVOLVE-BLOCK-END


class ModelParams:
    """
    Returns LightGBM parameters for the fraud detection model.
    
    Attributes:
        None
    """

    @staticmethod
    def get_params():
        params = {
            'objective': 'binary',
            'metric': 'auc',
            'boosting_type': 'gbdt',
            'num_leaves': 31,
            'learning_rate': 0.05,
            'feature_fraction': 0.85,
            'bagging_fraction': 0.85,
            'bagging_freq': 5,
            'max_depth': 6,
            'min_child_samples': 40,
            'scale_pos_weight': 15,
            'lambda_l1': 0.1,
            'lambda_l2': 1.0,
            'verbose': -1,
        }
        
        # EVOLVE-BLOCK-START
        params['num_leaves'] = np.random.randint(15, 63)
        params['learning_rate'] = np.random.uniform(0.01, 0.2)
        params['feature_fraction'] = np.random.uniform(0.5, 1.0)
        params['bagging_fraction'] = np.random.uniform(0.5, 1.0)
        params['min_child_samples'] = np.random.randint(10, 100)
        params['scale_pos_weight'] = np.random.randint(5, 25)
        params['lambda_l1'] = np.random.uniform(0.0, 5.0)
        params['lambda_l2'] = np.random.uniform(0.0, 5.0)
        
        return params
        # EVOLVE-BLOCK-END


class FusionScoreCalculator:
    """
    Combines transfer risk and account risk into a final suspicious-transfer score.
    
    Attributes:
        None
    """

    @staticmethod
    def compute_fusion_score(transfer_risk_scores, account_risk_scores):
        w_transfer = 0.80
        w_account = 0.20

        fused = w_transfer * transfer_risk_scores + w_account * account_risk_scores

        high_account_risk = account_risk_scores > 0.3
        fused = np.where(high_account_risk, np.clip(fused * 1.2, 0, 1), fused)

        threshold = float(np.percentile(fused, 92))

        return fused, threshold


def evaluate(seed=42):
    txn_df, acc_df = load_data(seed)
    
    # Early filtering of irrelevant columns for efficiency
    relevant_cols = [
        'amount_jpy', 'is_high_amount', 'balance_drain_ratio',
        'sender_account_age_days', 'sender_ib_enabled', 'receiver_is_new',
        'txn_count_24h', 'txn_volume_24h', 'has_recent_activity',
        'sender_mean_amount', 'sender_std_amount', 'sender_txn_count',
        'amount_zscore', 'is_weekend', 'is_late_night'
    ]
    
    txn_df = txn_df[relevant_cols + ['is_laundering']]

    # Feature engineering
    feature_engineer = FeatureEngineer(txn_df)
    evolved_feats = feature_engineer.engineer_features()

    # Combine base features and engineered features
    X_base = txn_df.drop(columns=['is_laundering'])
    X = pd.concat([X_base, evolved_feats], axis=1)
    
    y = txn_df['is_laundering']

    # Split data for training and evaluation
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, random_state=seed, stratify=y)

    # Get model parameters
    params = ModelParams.get_params()

    # Train LightGBM model
    train_data = lgb.Dataset(X_train, label=y_train)
    val_data = lgb.Dataset(X_test, label=y_test, reference=train_data)

    model = lgb.train(
        params,
        train_data,
        num_boost_round=200,
        valid_sets=[val_data],
        valid_names=['val'],
        callbacks=[
            lgb.early_stopping(stopping_rounds=20),
            lgb.log_evaluation(0)
        ]
    )

    # Predict
    y_pred_proba = model.predict(X_test)

    # Account risk scores for fusion
    account_risk_col = 'sender_account_risk_score'
    if account_risk_col in txn_df.columns:
        account_risk_scores = txn_df.loc[X_test.index, account_risk_col].fillna(0).values
    else:
        account_risk_scores = np.zeros(len(X_test))

    # Compute fusion score and threshold
    fused_scores, threshold = FusionScoreCalculator.compute_fusion_score(y_pred_proba, account_risk_scores)

    y_pred_binary = (fused_scores >= threshold).astype(int)
    
    metrics = {
        'roc_auc': roc_auc_score(y_test, fused_scores),
        'f1_score': f1_score(y_test, y_pred_binary, zero_division=0),
        'precision': precision_score(y_test, y_pred_binary, zero_division=0),
        'recall': recall_score(y_test, y_pred_binary, zero_division=0),
        'prevention_rate': int((y_pred_binary & y_test).sum()) / int(y_test.sum()),
        'alert_rate': float(y_pred_binary.mean())
    }

    # Alert quality and combined score calculations
    if metrics['alert_rate'] < 0.01 or metrics['alert_rate'] > 0.5:
        alert_quality = 0.0
    else:
        if 0.02 <= metrics['alert_rate'] <= 0.10:
            alert_quality = 1.0
        elif metrics['alert_rate'] < 0.02:
            alert_quality = metrics['alert_rate'] / 0.02
        else:
            alert_quality = max(0.0, 1.0 - (metrics['alert_rate'] - 0.10) / 0.40)

    combined_score = (
        0.40 * metrics['roc_auc'] +
        0.30 * metrics['f1_score'] +
        0.20 * metrics['prevention_rate'] +
        0.10 * alert_quality
    )

    metrics.update({
        'combined_score': combined_score,
        'alert_quality': alert_quality,
        'n_evolved_features': len(evolved_feats.columns),
        'threshold': float(threshold)
    })

    return metrics


if __name__ == '__main__':
    seed = int(sys.argv[1]) if len(sys.argv) > 1 else 42
    try:
        result = evaluate(seed=seed)
        print(json.dumps(result))
    except Exception as e:
        # Output zero metrics on failure
        print(json.dumps({
            'combined_score': 0.0,
            'error': str(e),
        }))
        sys.exit(1)
