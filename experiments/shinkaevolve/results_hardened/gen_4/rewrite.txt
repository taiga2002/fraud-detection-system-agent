# EVOLVE-BLOCK-START
def engineer_evolved_features(df):
    """
    Create advanced interaction features focusing on temporal patterns,
    velocity, and subtle risk indicators derived from weak signals.

    Returns a DataFrame with new feature columns only.
    """
    
    feats = pd.DataFrame(index=df.index)

    # Temporal pattern interactions
    # Late-night transactions combined with high amounts are often suspicious.
    feats['late_night_high_amount'] = df['is_late_night'].fillna(0) * df['is_very_high_amount'].fillna(0)
    
    # Weekend transfer patterns: fraudsters often exploit weekends for large transfers.
    feats['weekend_transfer'] = df['is_weekend'].fillna(0) * df['txn_count_24h'] / (df['sender_account_age_days'] + 1)

    # Velocity features
    # High velocity of transactions from suspicious devices is a strong indicator.
    feats['velocity_suspicious'] = (df['txn_volume_24h'].fillna(0) / df['amount_jpy']) * df['sender_suspicious_device']

    # New account + high amount interaction: new accounts receiving large amounts are highly suspicious.
    feats['new_acct_high_amount'] = df['receiver_is_new'].fillna(0) * df['is_high_amount'].fillna(0)

    # Risk score from weak signals
    # Ratio of transaction count to account age captures unusual activity patterns.
    feats['activity_ratio'] = (df['txn_count_24h'].fillna(0) + 1) / (df['sender_account_age_days'] + 1)
    
    # IB enablement timing relative to account creation can indicate fraudulent behavior.
    feats['ib_timing_risk'] = df['sender_ib_days_since_enabled'].fillna(9999) / (df['sender_account_age_days'] + 1)

    return feats

def compute_fusion_score(transfer_risk_scores, account_risk_scores):
    """
    Combine transfer risk and an inferred account risk score into a final suspicious-transfer score.
    
    Args:
        transfer_risk_scores: numpy array of P(fraud) from the LightGBM model.
        account_risk_scores: numpy array of sender_account_risk_score.

    Returns:
        fused_scores: numpy array of final risk scores (0-1).
        threshold: float, decision threshold for alerting.
    """
    
    # Weighted fusion with transfer risk dominating
    w_transfer = 0.95
    w_account = 0.05

    fused = w_transfer * transfer_risk_scores + w_account * account_risk_scores
    
    # Boost score when inferred IB timing risk is high
    ib_timing_high = np.clip(df['ib_timing_risk'].values, 0, 1) > 0.8
    if 'ib_timing_risk' in df.columns:
        fused = np.where(ib_timing_high, np.clip(fused * 1.3, 0, 1), fused)

    # Adaptive threshold based on score distribution percentile
    threshold = float(np.percentile(fused, 95))

    return fused, threshold
