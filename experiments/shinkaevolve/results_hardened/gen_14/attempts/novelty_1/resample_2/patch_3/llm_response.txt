def engineer_evolved_features(df):
    feats = pd.DataFrame(index=df.index)
    
    # Log-transformed amount features for better distribution handling
    feats['log_amount'] = np.log1p(df['amount_jpy'])
    
    # IB enablement ratio: how recently internet banking was enabled relative to account age
    feats['ib_enable_ratio'] = df['sender_ib_days_since_enabled'].fillna(9999) / (df['sender_account_age_days'] + 1)
    
    # Velocity intensity: volume per transaction in recent window
    feats['avg_recent_txn_amount'] = np.log(df['txn_volume_24h']) - np.log(df['txn_count_24h'] + 1)

    # New account + high amount interaction
    feats['new_acct_high_amount'] = df['receiver_is_new'].astype(int) * (df['is_high_amount'].fillna(0))**2
    
    # Suspicious device interaction with IB enablement time
    feats['susp_device_ib_enable'] = np.log(df['sender_suspicious_device'].fillna(1)) * feats['ib_enable_ratio']
    
    # Weekend transfer risk amplified by high volume/amount
    feats['weekend_transfer_volume_amount'] = (df['day_of_week'] >= 5).astype(int) * \
                                               ((np.log(df['txn_count_24h']) + np.log(df['amount_jpy'])) / np.sqrt(df['receiver_account_age_days'] + 1))

    # Interaction between high amount and suspicious device activity
    feats['high_amount_susp_device'] = df['is_high_amount'].fillna(0) * df['sender_suspicious_device'].fillna(0)

    return feats

def compute_fusion_score(transfer_risk_scores, account_risk_scores):
    w_transfer = 0.95 # Higher weight on transfer risk model output
    w_account = 0.05
    
    fused_scores = np.where(account_risk_scores > 0.3,
                            np.clip(w_transfer * (transfer_risk_scores ** 1.2) + w_account * account_risk_scores, 0, 1),
                            w_transfer * transfer_risk_scores + w_account * account_risk_scores)
    
    # Adaptive threshold: use percentile-based to handle varying score ranges
    threshold = np.percentile(fused_scores, 95)

    return fused_scores, threshold
