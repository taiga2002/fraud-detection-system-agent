def engineer_evolved_features(df):
    """
    Create interaction features from the base transaction columns.

    Available numeric columns (after leaky feature removal):
      amount_jpy, amount_log, receiver_account_age_days,
      sender_account_age_days, sender_ib_days_since_enabled,
      txn_count_24h, txn_volume_24h, amount_zscore, hour

    Available binary columns:
      is_high_amount, is_very_high_amount, first_transfer,
      sender_ib_enabled, sender_suspicious_device,
      receiver_is_new, receiver_is_very_new, has_recent_activity,
      amount_above_mean, amount_much_above_mean,
      is_weekend, is_business_hours, is_late_night

    NOTE: sender_mean_amount, sender_std_amount, sender_txn_count,
    balance_drain_ratio, sender_account_risk_score have been REMOVED
    (they were near-perfect proxies for the target). Feature engineering
    must create useful signals from the remaining weaker features.

    Returns a DataFrame with new feature columns only.
    """
    feats = pd.DataFrame(index=df.index)

    # Velocity intensity: volume per transaction in recent window
    feats['avg_recent_txn_amount'] = df['txn_volume_24h'].fillna(0) / (df['txn_count_24h'].fillna(1))
    
    # New account + high amount interaction
    feats['new_acct_high_amount'] = (df['receiver_is_new'] | df['receiver_is_very_new']) * df['is_high_amount']
    
    # Late night + high amount risk
    feats['late_night_high_amount'] = (df['is_late_night']) * df['is_high_amount']

    # First transfer to new receiver (classic scam pattern)
    feats['first_txn_new_receiver'] = df['first_transfer'] * df['receiver_is_new']
    
    # High velocity + suspicious device
    feats['velocity_suspicious'] = ((df['txn_count_24h'].fillna(0) > 5) & df['sender_suspicious_device']) + (df['txn_volume_24h'].fillna(0) / (df['txn_count_24h'].fillna(1)) > 100 * np.log(df['amount_jpy'] + 1))
    
    # IB enablement ratio: how recently IB was enabled relative to account age
    feats['ib_age_ratio'] = df['sender_ib_days_since_enabled'].fillna(999) / (df['sender_account_age_days'].fillna(0) + 1)
    
    # Temporal pattern interaction: late night and weekend transfers
    feats['late_night_weekend_transfer'] = df['is_late_night'] * df['is_weekend']
    
    return feats

def compute_fusion_score(transfer_risk_scores, account_risk_scores):
    """
    Combine transfer risk and account risk into a final suspicious-transfer score.

    Args:
        transfer_risk_scores: numpy array of P(fraud) from the LightGBM model
        account_risk_scores: numpy array of sender_account_risk_score

    Returns:
        fused_scores: numpy array of final risk scores (0-1)
        threshold: float, decision threshold for alerting
    """
    # Weighted fusion: transfer risk dominates, account risk amplifies
    w_transfer = 0.90
    w_account = 0.10

    fused = w_transfer * transfer_risk_scores + w_account * account_risk_scores
    
    # Boost fused score when account risk is elevated
    high_account_risk = account_risk_scores > 0.5
    fused[high_account_risk] *= 1.25

    # Adaptive threshold: use percentile-based to handle varying score ranges
    # Target ~5-8% alert rate (matching approximate fraud rate)
    threshold = float(np.percentile(fused, 95))

    return fused, threshold
