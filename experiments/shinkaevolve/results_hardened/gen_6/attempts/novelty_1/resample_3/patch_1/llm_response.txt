# EVOLVE-BLOCK-START
def engineer_evolved_features(df):
    feats = pd.DataFrame(index=df.index)

    # Log-transformed amount features
    feats['log_amount'] = np.log1p(df['amount_jpy'])
    
    # IB enablement ratio: how recently IB was enabled relative to account age
    feats['ib_age_ratio'] = df['sender_ib_days_since_enabled'].fillna(9999) / (df['sender_account_age_days'] + 1)

    # Velocity intensity: volume per transaction in recent window
    feats['avg_recent_txn_amount'] = df['txn_volume_24h'] / (df['txn_count_24h'] + 1)
    
    # New account + high amount interaction
    feats['new_acct_high_amount'] = df['receiver_is_new'].fillna(0) * df['is_high_amount'].fillna(0)

    # Late night + high amount risk
    feats['late_night_high_amount'] = df['is_late_night'].fillna(0) * df['is_high_amount'].fillna(0)
    
    # Weekend transfer risk
    feats['weekend_transfer'] = df['receiver_is_new'].fillna(0) * (df['day_of_week'] >= 5).astype(int)

    # High velocity + suspicious device
    feats['velocity_suspicious'] = (np.log1p(df['txn_count_24h']) + 1) * df['sender_suspicious_device'].fillna(0)
    
    return feats

def compute_fusion_score(transfer_risk_scores, account_risk_scores):
    w_transfer = 0.90 # Higher weight on transfer risk model output
    w_account = 0.10
    
    fused_scores = np.where(account_risk_scores > 0.3,
                            np.clip(w_transfer * transfer_risk_scores + w_account * (account_risk_scores ** 2), 0, 1),
                            w_transfer * transfer_risk_scores + w_account * account_risk_scores)
    
    # Adaptive threshold: use percentile-based to handle varying score ranges
    threshold = np.percentile(fused_scores, 95) if np.any(account_risk_scores > 0.3) else np.percentile(fused_scores, 92)

    return fused_scores, threshold
