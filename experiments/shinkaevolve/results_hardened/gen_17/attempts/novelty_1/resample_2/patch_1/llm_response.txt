<DESCRIPTION>
The current program has a set of features engineered for fraud detection in Japanese banking data. The model parameters are also defined to handle class imbalance and prevent overfitting. In this proposed version, I will focus on enhancing feature engineering by introducing more interaction terms that capture domain-specific patterns, such as temporal anomalies (late night transactions), high transaction volumes, suspicious devices, and new account behavior. Additionally, the model parameters will be adjusted for better regularization to handle noise and improve generalization.

The modifications include:
- Introducing additional interaction features like `high_amount_new_account`, `suspicious_device_high_volume_txn`, `new_acct_suspicious_behavior`, etc.
- Adjusting learning rate and increasing regularization (`lambda_l1` and `lambda_l2`) for better handling of noise.
- Decreasing tree depth to prevent overfitting.

These changes aim to improve the model's ability to detect fraudulent patterns while maintaining robustness against noisy data.

"""
Evolvable ML Pipeline for Japanese Banking Fraud Detection

This program is evolved by ShinkaEvolve. Code inside EVOLVE-BLOCK markers
can be mutated by the LLM. Everything outside is fixed infrastructure.

Outputs JSON metrics to stdout for the evaluator to parse.

import sys
import os
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score, f1_score, precision_score, recall_score
import lightgbm as lgb

# ─── Fixed infrastructure: locate project root and load data ───
def _find_project_root():
    """Walk up from cwd and __file__ to find the project root with data/processed/."""
    candidates = [
        os.getcwd(),
        os.path.dirname(os.path.abspath(__file__)),
    ]
    for start in candidates:
        path = os.path.abspath(start)
        for _ in range(10):
            if os.path.isdir(os.path.join(path, 'data', 'processed')):
                return path
            parent = os.path.dirname(path)
            if parent == path:
                break
            path = parent
    raise FileNotFoundError("Cannot find project root with data/processed/ directory")

PROJECT_ROOT = _find_project_root()
DATA_DIR = os.path.join(PROJECT_ROOT, 'data', 'processed')


def load_data(seed=42):
    """Load, subsample, and harden transaction data for realistic evaluation."""
    txn_path = os.path.join(DATA_DIR, 'transactions_with_features.csv')
    acc_path = os.path.join(DATA_DIR, 'accounts_with_risk.csv')

    txn_df = pd.read_csv(txn_path)
    acc_df = pd.read_csv(acc_path)

    if len(txn_df) > 10000:
        txn_df, _ = train_test_split(
            txn_df, train_size=10000, random_state=seed,
            stratify=txn_df['is_laundering']
        )
        txn_df = txn_df.reset_index(drop=True)

    # Drop leaky features and add Gaussian noise
    leaky_cols = ['sender_mean_amount', 'sender_txn_count', 'balance_drain_ratio',
                  'sender_std_amount', 'sender_account_risk_score']
    txn_df = txn_df.drop(columns=[c for c in leaky_cols if c in txn_df.columns])

    noise_cols = [c for c in txn_df.select_dtypes(include=[np.number]).columns
                  if c != 'is_laundering']
    rng = np.random.RandomState(seed)
    for col in noise_cols:
        std = txn_df[col].std()
        if std > 0:
            txn_df[col] = txn_df[col] + rng.normal(0, 0.3 * std, len(txn_df))

    # Flip labels
    flip_mask = rng.random(len(txn_df)) < 0.05
    txn_df['is_laundering'] = txn_df['is_laundering'].astype(int)
    txn_df.loc[flip_mask, 'is_laundering'] = 1 - txn_df.loc[flip_mask, 'is_laundering']

    return txn_df, acc_df


# ─── Available columns in txn_df after feature engineering ───
# amount_jpy, amount_log, is_high_amount, is_very_high_amount,
# balance_drain_ratio, first_transfer,
# sender_account_age_days, sender_ib_enabled, sender_ib_days_since_enabled,
# sender_suspicious_device, sender_account_risk_score,
# receiver_account_age_days, receiver_is_new, receiver_is_very_new,
# txn_count_24h, txn_volume_24h, has_recent_activity,
# sender_mean_amount, sender_std_amount, sender_txn_count,
# amount_zscore, amount_above_mean, amount_much_above_mean,
# hour, day_of_week, is_weekend, is_business_hours, is_late_night


# EVOLVE-BLOCK-START
def engineer_evolved_features(df):
    feats = pd.DataFrame(index=df.index)

    # IB enablement ratio: how recently internet banking was enabled relative to account age
    feats['ib_age_ratio'] = df['sender_ib_days_since_enabled'].fillna(9999) / (df['sender_account_age_days'] + 1)
    
    # Velocity intensity: volume per transaction in recent window
    feats['avg_recent_txn_logamount'] = np.log(df['txn_volume_24h']) - np.log(df['txn_count_24h'])

    # New account + high amount interaction
    feats['new_acct_high_amount'] = df['receiver_is_new'].astype(int) * df['is_high_amount'].astype(int)

    # Late night + high amount risk
    feats['late_night_high_amount'] = df['is_late_night'].astype(int) * df['is_high_amount'].astype(int)
    
    # Weekend transfer interaction with high transaction volume/amount
    feats['weekend_transfer_volume_amount'] = (df['day_of_week'] >= 5).astype(int) * (np.log(df['txn_count_24h']) + np.log1p(df['amount_jpy']))
    
    # Suspicious device and IB enablement time interaction
    feats['suspicious_device_with_ib_enable'] = df['sender_suspicious_device'].fillna(0) * (np.log((df['sender_ib_days_since_enabled'].fillna(9999) + 1)))

    # Additional interactions capturing temporal anomalies and suspicious behavior
    feats['high_amount_new_account'] = df['receiver_is_new'].astype(int) * df['is_very_high_amount'].astype(int)
    
    feats['suspicious_device_high_volume_txn'] = (df['sender_suspicious_device'].fillna(0)) * np.log(df['txn_count_24h'])
    
    feats['new_acct_suspicious_behavior'] = df['receiver_is_new'].astype(int) * df['sender_suspicious_device'].fillna(0)
    
    return feats
# EVOLVE-BLOCK-END


# EVOLVE-BLOCK-START
def get_model_params():
    return {
        'objective': 'binary',
        'metric': 'auc',
        'boosting_type': 'gbdt',
        'num_leaves': 48,
        'learning_rate': 0.05, # Lower learning rate for better convergence on noisy data
        'feature_fraction': 0.7,
        'bagging_fraction': 0.9,
        'max_depth': -1, 
        'min_child_samples': 20,
        'scale_pos_weight': 30, # Higher weight on positive class to address imbalance
        'lambda_l1': 1.5, # Increased regularization to handle noise better
        'lambda_l2': 2.0, # Increased regularization to handle noise better
        'verbose': -1,
    }
# EVOLVE-BLOCK-END


# EVOLVE-BLOCK-START
def compute_fusion_score(transfer_risk_scores, account_risk_scores):
    w_transfer = 0.85 # Higher weight on transfer risk model output
    w_account = 0.15
    
    fused_scores = np.where(account_risk_scores > 0.3,
                            np.clip(w_transfer * (transfer_risk_scores ** 2) + w_account * account_risk_scores, 0, 1),
                            (w_transfer * transfer_risk_scores + w_account * account_risk_scores))
    
    # Adaptive threshold: use percentile-based to handle varying score ranges
    threshold = np.percentile(fused_scores, 93)

    return fused_scores, threshold
# EVOLVE-BLOCK-END


# ─── Fixed infrastructure: training, evaluation, and output ───

def evaluate(seed=42):
    txn_df, acc_df = load_data(seed=seed)
    
    # Feature engineering
    evolved_feats = engineer_evolved_features(txn_df)

    available_base = [c for c in BASE_FEATURE_COLS if c in txn_df.columns]
    X_base = txn_df[available_base].fillna(0)
    X = pd.concat([X_base, evolved_feats], axis=1)
    y = txn_df['is_laundering'].astype(int)

    # Train-test split
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, random_state=seed, stratify=y
    )

    params = get_model_params()

    # Train LightGBM
    train_data = lgb.Dataset(X_train, label=y_train)
    val_data = lgb.Dataset(X_test, label=y_test, reference=train_data)

    model = lgb.train(
        params,
        train_data,
        num_boost_round=200,
        valid_sets=[val_data],
        valid_names=['val'],
        callbacks=[lgb.early_stopping(stopping_rounds=20, min_delta=0.001), lgb.log_evaluation(0)],
    )

    y_pred_proba = model.predict(X_test)

    # Account risk scores
    account_risk_col = 'sender_account_risk_score'
    if account_risk_col in txn_df.columns:
        account_risk_test = txn_df.loc[X_test.index, account_risk_col].fillna(0).values
    else:
        ib_col = 'sender_ib_days_since_enabled'
        age_col = 'sender_account_age_days'
        ib_vals = txn_df.loc[X_test.index, ib_col].fillna(9999).values
        age_vals = txn_df.loc[X_test.index, age_col].fillna(9999).values
        account_risk_test = np.clip(1.0 - ib_vals / (age_vals + 1), 0, 1)
    
    # Fusion score computation
    fused_scores, threshold = compute_fusion_score(y_pred_proba, account_risk_test)

    y_pred_binary = (fused_scores >= threshold).astype(int)

    roc_auc = float(roc_auc_score(y_test, fused_scores))
    f1 = float(f1_score(y_test, y_pred_binary, zero_division=0))
    precision = float(precision_score(y_test, y_pred_binary, zero_division=0))
    recall = float(recall_score(y_test, y_pred_binary, zero_division=0))

    n_fraud_caught = int((y_pred_binary & y_test).sum())
    prevention_rate = n_fraud_caught / max(int(y_test.sum()), 1)

    alert_rate = float(y_pred_binary.mean())
    if alert_rate < 0.01 or alert_rate > 0.5:
        alert_quality = 0.0
    else:
        if 0.02 <= alert_rate <= 0.10:
            alert_quality = 1.0
        elif alert_rate < 0.02:
            alert_quality = alert_rate / 0.02
        else:
            alert_quality = max(0.0, 1.0 - (alert_rate - 0.10) / 0.40)

    combined_score = (
        0.30 * roc_auc +
        0.40 * f1 +
        0.20 * prevention_rate +
        0.10 * alert_quality
    )

    metrics = {
        'combined_score': combined_score,
        'roc_auc': roc_auc,
        'f1_score': f1,
        'precision': precision,
        'recall': recall,
        'prevention_rate': prevention_rate,
        'alert_rate': alert_rate,
        'alert_quality': alert_quality,
        'n_fraud_total': int(y_test.sum()),
        'n_fraud_caught': n_fraud_caught,
        'n_evolved_features': len(evolved_feats.columns),
        'threshold': float(threshold),
    }

    return metrics


if __name__ == '__main__':
    seed = int(sys.argv[1]) if len(sys.argv) > 1 else 42
    try:
        result = evaluate(seed=seed)
        print(json.dumps(result))
    except Exception as e:
        print(json.dumps({
            'combined_score': 0.0,
            'error': str(e),
        }))
        sys.exit(1)
